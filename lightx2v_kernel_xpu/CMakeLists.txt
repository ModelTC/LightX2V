cmake_minimum_required(VERSION 3.22 FATAL_ERROR)
project(sycl-kernels LANGUAGES CXX)

# Keep this opt-in because it requires an external CUTLASS-SYCL/sycl-tla tree.
# It must remain disabled on Windows: the current sycl-tla FMHA kernel builds
# there, but produces incorrect attention results.
set(ENABLE_CUTE_FMHA_DEFAULT OFF)
option(ENABLE_CUTE_FMHA "Build the Linux-only CUTLASS-SYCL CUTE FMHA library"
    ${ENABLE_CUTE_FMHA_DEFAULT})
set(CUTLASS_SYCL_ROOT "$ENV{CUTLASS_SYCL_ROOT}" CACHE PATH
    "Optional path to a local CUTLASS-SYCL/sycl-tla source tree")
set(SYCL_TLA_GIT_REPOSITORY "https://github.com/intel/sycl-tla.git" CACHE STRING
    "sycl-tla repository used when CUTLASS_SYCL_ROOT is not set")
set(SYCL_TLA_GIT_REVISION "2fc09973bfdf15755090fcb0e3b6ad236408a992" CACHE STRING
    "Pinned sycl-tla revision used for CUTE FMHA")
set(XPU_TARGET "bmg" CACHE STRING "Intel GPU AOT target (bmg or ptl-h)")

find_package(IntelSYCL REQUIRED)

# Python (scikit-build-core injects paths; this finds Interpreter + Development.Module)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)

# ── ur_api.h shim ─────────────────────────────────────────────────────────────
# conda puts an old ur_api.h that lacks symbols required by oneAPI 2025.x.
# FindIntelSYCL (called above) already resolves the 'latest' junction and sets
# SYCL_INCLUDE_DIR to the real versioned path (e.g. .../compiler/2025.2/include).
# Use that directly so the shim never embeds the unresolvable 'latest' symlink.
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/includes")
file(WRITE "${CMAKE_BINARY_DIR}/includes/ur_api.h"
    "#pragma once\n// Shim: redirect to the compiler-matched ur_api.h\n// Generated by CMake -- do not edit by hand.\n#include \"${SYCL_INCLUDE_DIR}/ur_api.h\"\n"
)
if(WIN32)
    set(CMAKE_CXX_FLAGS "/I\"${CMAKE_BINARY_DIR}/includes\" ${CMAKE_CXX_FLAGS}")
else()
    add_compile_options("-I${CMAKE_BINARY_DIR}/includes")
endif()
message(STATUS "ur_api.h shim → ${SYCL_INCLUDE_DIR}/ur_api.h")

find_package(Torch REQUIRED)

if(XPU_TARGET STREQUAL "ptl-h")
    set(XPU_ARCH_DEFINE OMNI_XPU_ARCH_PTL_H=1)
elseif(XPU_TARGET STREQUAL "bmg")
    set(XPU_ARCH_DEFINE OMNI_XPU_ARCH_BMG=1)
else()
    message(FATAL_ERROR "Unsupported XPU_TARGET=${XPU_TARGET}; use ptl-h or bmg")
endif()

set(MODULE_NAME _ext)

target_compile_definitions(${MODULE_NAME} PRIVATE TORCH_EXTENSION_NAME=${MODULE_NAME})

add_sycl_to_target(TARGET ${MODULE_NAME} SOURCES ${SYCL_SRCS})

if(WIN32)
    target_compile_options(${MODULE_NAME} PRIVATE -Wno-ignored-attributes)
endif()

# Force oneAPI compiler headers before conda's conflicting headers.
# SYCL_INCLUDE_DIR is the real versioned path set by FindIntelSYCL.
if(CMAKE_CXX_COMPILER_ID MATCHES "IntelLLVM")
    target_include_directories(${MODULE_NAME} BEFORE PRIVATE
        "${SYCL_INCLUDE_DIR}/sycl"
        "${SYCL_INCLUDE_DIR}"
    )
endif()

target_include_directories(${MODULE_NAME} PRIVATE "${CMAKE_BINARY_DIR}/includes")
target_include_directories(${MODULE_NAME} PRIVATE ${TORCH_INCLUDE_DIRS})

# oneDNN library (from active conda env or oneAPI install)
execute_process(
    COMMAND ${Python_EXECUTABLE} -c "import sys; print(sys.prefix)"
    OUTPUT_VARIABLE Python_PREFIX
    OUTPUT_STRIP_TRAILING_WHITESPACE
)
set(DNNL_LIBRARY_HINTS
    "${Python_PREFIX}/Library/lib"
    "${Python_PREFIX}/lib"
    "${Python_PREFIX}/lib64"
    "$ENV{ONEAPI_ROOT}/dnnl/latest/lib"
)
if(WIN32)
    list(APPEND DNNL_LIBRARY_HINTS
        "C:/Program Files (x86)/Intel/oneAPI/dnnl/latest/lib"
        "C:/Program Files/Intel/oneAPI/dnnl/latest/lib"
    )
endif()
find_library(DNNL_LIBRARY
    NAMES dnnl
    PATHS ${DNNL_LIBRARY_HINTS}
    NO_DEFAULT_PATH
)
if(NOT DNNL_LIBRARY)
    message(FATAL_ERROR
        "oneDNN (dnnl) not found under ${Python_PREFIX}.\n"
        "Activate the lightx2v_kernel conda env before building.")
endif()
set(DNNL_INCLUDE_HINTS
    "${Python_PREFIX}/include"
    "$ENV{ONEAPI_ROOT}/dnnl/latest/include"
)
if(WIN32)
    list(APPEND DNNL_INCLUDE_HINTS
        "C:/Program Files (x86)/Intel/oneAPI/dnnl/latest/include"
        "C:/Program Files/Intel/oneAPI/dnnl/latest/include"
    )
endif()
find_path(DNNL_INCLUDE_DIR
    NAMES oneapi/dnnl/dnnl.hpp
    PATHS ${DNNL_INCLUDE_HINTS}
    NO_DEFAULT_PATH
)
if(NOT DNNL_INCLUDE_DIR)
    message(FATAL_ERROR
        "oneDNN headers not found for ${DNNL_LIBRARY}.\n"
        "Install matching oneDNN development headers under ${Python_PREFIX}.")
endif()
message(STATUS "Found oneDNN: ${DNNL_LIBRARY}")
message(STATUS "Found oneDNN headers: ${DNNL_INCLUDE_DIR}")

# Torch may ship a different oneDNN version from the separately linked runtime.
# Wrap the complete oneDNN public header tree so transitive includes cannot fall
# back to Torch's headers. Adding the whole prefix first could also override the
# compiler's SYCL headers.
set(DNNL_HEADER_ROOT "${DNNL_INCLUDE_DIR}/oneapi/dnnl")
set(DNNL_SHIM_ROOT "${CMAKE_BINARY_DIR}/dnnl_include")
file(GLOB_RECURSE DNNL_PUBLIC_HEADERS
    RELATIVE "${DNNL_INCLUDE_DIR}"
    "${DNNL_HEADER_ROOT}/*.h"
    "${DNNL_HEADER_ROOT}/*.hpp"
)
foreach(DNNL_HEADER IN LISTS DNNL_PUBLIC_HEADERS)
    get_filename_component(DNNL_HEADER_DIR "${DNNL_HEADER}" DIRECTORY)
    file(MAKE_DIRECTORY "${DNNL_SHIM_ROOT}/${DNNL_HEADER_DIR}")
    file(TO_CMAKE_PATH
        "${DNNL_INCLUDE_DIR}/${DNNL_HEADER}" DNNL_HEADER_SOURCE)
    file(WRITE "${DNNL_SHIM_ROOT}/${DNNL_HEADER}"
        "#pragma once\n#include \"${DNNL_HEADER_SOURCE}\"\n")
endforeach()
target_include_directories(${MODULE_NAME} BEFORE PRIVATE
    "${DNNL_SHIM_ROOT}")

# ESIMD SDP library (must be built before configuring this project).
# Windows links against the DLL import library; Linux links against the shared
# object and resolves it beside the Python extension at runtime.
add_library(esimd_lgrf_lib SHARED IMPORTED)
if(WIN32)
    set_target_properties(esimd_lgrf_lib PROPERTIES
        IMPORTED_IMPLIB "${CMAKE_CURRENT_SOURCE_DIR}/lgrf_uni/esimd.unify.lgrf.lib"
        IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/lgrf_uni/esimd.unify.lgrf.dll")
    set(ESIMD_RUNTIME "lgrf_uni/esimd.unify.lgrf.dll")
else()
    set_target_properties(esimd_lgrf_lib PROPERTIES
        IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/lgrf_uni/libesimd.unify.lgrf.so")
    set_target_properties(${MODULE_NAME} PROPERTIES
        BUILD_RPATH "\$ORIGIN"
        INSTALL_RPATH "\$ORIGIN")
    set(ESIMD_RUNTIME "lgrf_uni/libesimd.unify.lgrf.so")
endif()

find_library(TORCH_PYTHON_LIBRARY torch_python PATH "${TORCH_INSTALL_PREFIX}/lib")

target_link_libraries(${MODULE_NAME} PRIVATE
    ${TORCH_LIBRARIES}
    ${TORCH_PYTHON_LIBRARY}
    ${DNNL_LIBRARY}
    esimd_lgrf_lib
)

# Keep RMSNorm independent from the legacy extension and its oneDNN ABI. This
# sidecar only links PyTorch/SYCL and is loaded lazily through torch.ops.
add_library(rms_norm_torch SHARED csrc/rms_norm.cpp)
set_target_properties(rms_norm_torch PROPERTIES
    PREFIX ""
    BUILD_RPATH "${TORCH_INSTALL_PREFIX}/lib"
    INSTALL_RPATH "\$ORIGIN;${TORCH_INSTALL_PREFIX}/lib")
target_compile_features(rms_norm_torch PRIVATE cxx_std_17)
target_include_directories(rms_norm_torch PRIVATE
    ${Python_INCLUDE_DIRS}
    ${TORCH_INCLUDE_DIRS})
add_sycl_to_target(TARGET rms_norm_torch SOURCES csrc/rms_norm.cpp)
target_link_libraries(rms_norm_torch PRIVATE
    ${TORCH_LIBRARIES}
    ${TORCH_PYTHON_LIBRARY})

if(ENABLE_CUTE_FMHA)
    if(WIN32)
        message(FATAL_ERROR
            "CUTE FMHA is disabled on Windows because sycl-tla produces incorrect results")
    endif()
    if(NOT CUTLASS_SYCL_ROOT)
        include(FetchContent)
        message(STATUS
            "CUTLASS_SYCL_ROOT is not set; fetching sycl-tla ${SYCL_TLA_GIT_REVISION}")
        FetchContent_Declare(sycl_tla
            GIT_REPOSITORY ${SYCL_TLA_GIT_REPOSITORY}
            GIT_TAG ${SYCL_TLA_GIT_REVISION}
            GIT_PROGRESS TRUE
            GIT_SHALLOW FALSE)
        FetchContent_GetProperties(sycl_tla)
        if(NOT sycl_tla_POPULATED)
            FetchContent_Populate(sycl_tla)
        endif()
        set(CUTLASS_SYCL_ROOT "${sycl_tla_SOURCE_DIR}")
    else()
        message(STATUS "Using local sycl-tla source: ${CUTLASS_SYCL_ROOT}")
    endif()
    foreach(required_dir include tools/util/include examples/common applications)
        if(NOT IS_DIRECTORY "${CUTLASS_SYCL_ROOT}/${required_dir}")
            message(FATAL_ERROR
                "CUTE FMHA requires CUTLASS_SYCL_ROOT with ${required_dir}; "
                "got '${CUTLASS_SYCL_ROOT}'")
        endif()
    endforeach()

    add_library(cute_fmha_torch SHARED cute/cute_fmha_torch.cpp)
    set_target_properties(cute_fmha_torch PROPERTIES
        PREFIX ""
        BUILD_RPATH "${TORCH_INSTALL_PREFIX}/lib"
        INSTALL_RPATH "\$ORIGIN;${TORCH_INSTALL_PREFIX}/lib")
    target_compile_features(cute_fmha_torch PRIVATE cxx_std_17)
    target_compile_definitions(cute_fmha_torch PRIVATE
        CUTLASS_ENABLE_SYCL
        SYCL_INTEL_TARGET
        ${XPU_ARCH_DEFINE})
    target_include_directories(cute_fmha_torch PRIVATE
        "${CUTLASS_SYCL_ROOT}/include"
        "${CUTLASS_SYCL_ROOT}/tools/util/include"
        "${CUTLASS_SYCL_ROOT}/examples/common"
        "${CUTLASS_SYCL_ROOT}/applications"
        "${CMAKE_CURRENT_SOURCE_DIR}/cute"
        ${TORCH_INCLUDE_DIRS})
    target_compile_options(cute_fmha_torch PRIVATE
        -O3
        -fsycl
        -fsycl-targets=spir64_gen
        -Xsycl-target-backend "-device ${XPU_TARGET}"
        -fno-sycl-instrument-device-code
        -Wno-unknown-pragmas
        -Wno-unused-variable
        -Wno-unused-but-set-variable
        -Wno-unused-local-typedef
        -Wno-uninitialized
        -Wno-reorder-ctor
        -Wno-logical-op-parentheses
        -Wno-unused-function
        -Wno-deprecated-copy)
    target_link_options(cute_fmha_torch PRIVATE
        -fsycl
        -fsycl-targets=spir64_gen
        -Xsycl-target-backend "-device ${XPU_TARGET}"
        -Xspirv-translator
        "-spirv-ext=+SPV_INTEL_split_barrier,+SPV_INTEL_2d_block_io,+SPV_INTEL_subgroup_matrix_multiply_accumulate")
    target_link_libraries(cute_fmha_torch PRIVATE
        ${TORCH_LIBRARIES}
        ${TORCH_PYTHON_LIBRARY})
endif()
